cmake_minimum_required(VERSION 3.24)
project(rudp VERSION 1.0.0 DESCRIPTION "Reliable byte stream over UDP" LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(RUDP_BUILD_TESTS "Build unit tests" ON)
option(RUDP_BUILD_APPS "Build command line tools" ON)
option(RUDP_WERROR "Treat warnings as errors" OFF)
option(RUDP_SANITIZE "AddressSanitizer + UBSan (GCC/Clang)" OFF)

add_library(rudp_warnings INTERFACE)
if(MSVC AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  target_compile_options(rudp_warnings INTERFACE /W4 /permissive- /utf-8 $<$<BOOL:${RUDP_WERROR}>:/WX>)
else()
  target_compile_options(rudp_warnings INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wold-style-cast
    $<$<BOOL:${RUDP_WERROR}>:-Werror>)
endif()
if(RUDP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

add_library(rudp src/packet.cpp src/connection.cpp src/sim.cpp src/udp.cpp)
add_library(rudp::rudp ALIAS rudp)
target_include_directories(rudp PUBLIC include)
target_link_libraries(rudp PRIVATE rudp_warnings)
if(WIN32)
  target_link_libraries(rudp PUBLIC ws2_32)
  target_compile_definitions(rudp PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
find_package(Threads REQUIRED)
target_link_libraries(rudp PUBLIC Threads::Threads)

if(RUDP_BUILD_APPS)
  foreach(app rudp_sim rudp_bench rudp_send rudp_recv)
    add_executable(${app} apps/${app}.cpp)
    target_link_libraries(${app} PRIVATE rudp rudp_warnings)
  endforeach()
endif()

if(RUDP_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
